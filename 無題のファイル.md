function doPost(e) {

const contents = JSON.parse(e.postData.contents);

const action = contents.action;

const ss = SpreadsheetApp.getActiveSpreadsheet();

const sheet = ss.getSheetByName(SHEET_NAME);

function bulkRecordEntries(entries, sheet) {

const results = [];

const budgetSummaries = {};

  

entries.forEach(data => {

const dateStr = data.date || Utilities.formatDate(new Date(), "JST", "yyyy/MM/dd");

const month = parseInt(dateStr.split('/')[1]).toString();

const account = data.account || "三井住友";

const type = data.type || "出金";

const item1 = data.category1 || "変動費";

const item2 = data.item;

const amount = data.amount;

const lastRow = sheet.getLastRow();

const formulaH = `=IFERROR(INDEX(H:H,MAX(FILTER(ROW($C$1:C${lastRow}),$C$1:C${lastRow}="${account}"))),0) + IF(D${lastRow+1}="入金",G${lastRow+1},-G${lastRow+1})`;

sheet.appendRow([month, dateStr, account, type, item1, item2, amount, formulaH]);

// 予算チェック（同じ項目は一度だけ計算）

if (!budgetSummaries[item2]) {

budgetSummaries[item2] = getBudgetStatus(item2, month);

}

results.push({item: item2, amount: amount, status: 'success'});

});

return ContentService.createTextOutput(JSON.stringify({

status: 'success',

processed_count: entries.length,

details: results,

budgets: budgetSummaries

})).setMimeType(ContentService.MimeType.JSON);

  

function getBudgetStatus(itemName, monthStr) {

const ss = SpreadsheetApp.getActiveSpreadsheet();

const budgetSheet = ss.getSheetByName("予算");

const month = parseInt(monthStr);

const data = budgetSheet.getDataRange().getValues();

// 4行目が月（1月, 2月...）、B列（インデックス1）が項目名

let monthCol = -1;

for (let j = 0; j < data[3].length; j++) {

if (data[3][j] == month + "月") { monthCol = j; break; }

}

if (monthCol === -1) return null;

for (let i = 4; i < data.length; i++) {

if (data[i][1] == itemName) { // B列と一致するか

return {

item: itemName,

budget: data[i][monthCol],

current: data[i][monthCol + 1], // 実績列

remaining: data[i][monthCol] - data[i][monthCol + 1]

};

}

}

return null;

}
